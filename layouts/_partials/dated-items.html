{{- /* Renders a reverse-chronological list of reference items (publications, talks, projects). These share nearly identical shape, so all three sections render through this one partial. Front matter `description`, if set, renders immediately under the title. The date is omitted entirely when a page has none, so undated entries (long-running projects) render cleanly. @param {page.Pages} pages Pages to render, already sorted. @param {[]string} metaKeys Front-matter keys joined into the meta line, in order. e.g. (slice "venue") for publications, (slice "event" "location") for talks. @param {string} dateFormat Optional Go time layout. Defaults to year only. The title links to the item's own page only when that page has body content (an abstract, notes, a transcript). Items that are just a citation and a set of links render as plain text, so adding one never creates an empty page. @example {{ partial "dated-items.html" (dict "pages" .Pages "metaKeys" (slice "venue")) }} */}} {{- $metaKeys := .metaKeys | default slice }} {{- $dateFormat := .dateFormat | default "2006" }} {{- with .pages }}
    {{- range $page := . }}
  1. {{- if $page.Content -}} {{ $page.Title }} {{- else -}} {{ $page.Title }} {{- end -}} {{- if not $page.Date.IsZero }} {{- $page.Date.Format $dateFormat -}} {{- end }}
    {{- with $page.Description }}
    {{ . }}
    {{- end }} {{- with $page.Params.authors }}
    {{- partial "authors.html" (dict "authors" .) -}}
    {{- end }} {{- $meta := slice }} {{- range $k := $metaKeys }} {{- with index $page.Params $k }} {{- $meta = $meta | append . }} {{- end }} {{- end }} {{- with $meta }}
    {{ delimit . " · " }}
    {{- end }} {{- with $page.Params.links }}
    {{- range . }} {{ .label }} {{- end }}
    {{- end }}
    2. {{- end }}
{{- end }}
